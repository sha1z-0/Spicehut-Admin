static Future<PermissionStatus> requestAllPermissions(
  BuildContext context,
) async {
  try {
    // Show explanation dialog first
    await DialogUtils.showAnimatedDialog(
      context: context,
      barrierDismissible: false,
      builder: (ctx) => AlertDialog(
        title: const Text('Permissions Needed'),
        content: SingleChildScrollView(
          child: Column(
            mainAxisSize: MainAxisSize.min,
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              const Text(
                'This app needs the following permissions to function properly:',
                style: TextStyle(fontWeight: FontWeight.bold),
              ),
              const SizedBox(height: 16),
              _buildPermissionItem(
                icon: Icons.photo_camera,
                title: 'Camera',
                reason: 'To capture photos and scan items for order management',
              ),
              _buildPermissionItem(
                icon: Icons.storage,
                title: 'Storage',
                reason: 'To save documents, receipts, and order records on your device',
              ),
            ],
          ),
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(ctx),
            child: const Text('Cancel'),
          ),
          ElevatedButton(
            style: ElevatedButton.styleFrom(
              backgroundColor: const Color(0xFFFF7A00),
            ),
            onPressed: () => Navigator.pop(ctx),
            child: const Text(
              'Grant Permissions',
              style: TextStyle(color: Colors.white),
            ),
          ),
        ],
      ),
    );

    // List of permissions to request (excluding INTERNET which is built-in)
    final permissions = [
      Permission.camera,
      Permission.storage,
    ];

    // Request all permissions at once
    final Map<Permission, PermissionStatus> statuses =
        await permissions.request();

    // Separate results
    bool allGranted = true;
    List<String> deniedPermissions = [];

    statuses.forEach((permission, status) {
      if (status.isDenied) {
        deniedPermissions.add(_getPermissionName(permission));
        allGranted = false;
      } else if (status.isPermanentlyDenied) {
        deniedPermissions.add('${_getPermissionName(permission)} (permanently)');
        allGranted = false;
      }
    });

    // Show result to user
    if (!allGranted && deniedPermissions.isNotEmpty) {
      await DialogUtils.showErrorDialog(
        context: context,
        title: 'Permissions Requested',
        message:
            'Some permissions were not granted:\n\n${deniedPermissions.join('\n')}\n\nYou can enable them later in Settings.',
        buttonText: 'OK',
      );
    } else if (allGranted) {
      await DialogUtils.showSuccessDialog(
        context: context,
        title: 'Permissions Granted',
        message:
            'All necessary permissions have been granted. The app is ready to use!',
        buttonText: 'Continue',
      );
    }

    return statuses.values.first;
  } catch (e) {
    await DialogUtils.showErrorDialog(
      context: context,
      title: 'Permission Error',
      message: 'An error occurred while requesting permissions: $e',
      buttonText: 'OK',
    );
    return PermissionStatus.denied;
  }
}
