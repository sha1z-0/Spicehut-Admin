static Future<List<PrinterDevice>> scanLanPrinters({int port = 9100}) async {
  final devices = <PrinterDevice>[];

  try {
    // Get local IP to determine subnet
    final interfaces = await NetworkInterface.list();
    for (final interface in interfaces) {
      for (final addr in interface.addresses) {
        if (addr.type == InternetAddressType.IPv4) {
          final ip = addr.address;
          // Extract subnet (e.g., "192.168.1" from "192.168.1.100")
          final parts = ip.split('.');
          if (parts.length == 4) {
            final subnet = parts.take(3).join('.');

            // Try IPs 1-20 and 240-254 in the subnet (common printer range)
            final rangesToCheck = [
              ...List.generate(20, (i) => i + 1),
              ...List.generate(15, (i) => 240 + i),
            ];

            for (final i in rangesToCheck) {
              final testIp = '$subnet.$i';
              final reachable = await _canConnect(testIp, port, timeout: const Duration(milliseconds: 300));
              if (reachable) {
                devices.add(
                  PrinterDevice(
                    name: 'LAN Printer ($testIp)',
                    address: testIp,
                    port: port,
                    type: PrinterConnectionType.lan,
                  ),
                );
              }
            }
          }
        }
      }
    }
  } catch (_) {
    // Network scanning not available
  }

  return devices;
}
