static Future<void> printCustomerBill(
  String orderId,
  double totalAmount, {
  List<Map<String, dynamic>>? items,
  String? customerName,
  double? taxAmount,
  double? discountAmount,
  double? tip,
}) async {
  _ensureConnected();

  final lines = <String>[
    'SPICE HUT - BILL',
    '================================',
    'Order #: $orderId',
    if (customerName != null && customerName.isNotEmpty) 'Customer: $customerName',
    'Date: ${DateTime.now()}',
    '--------------------------------',
    'ITEMS:',
  ];

  double subtotal = 0;
  if (items != null && items.isNotEmpty) {
    for (final item in items) {
      final name = item['name'] ?? 'Item';
      final qty = item['quantity'] ?? 1;
      final price = (item['price'] ?? 0).toDouble();
      final itemTotal = qty * price;
      subtotal += itemTotal;
      lines.add('$qty x $name  CAD ${itemTotal.toStringAsFixed(2)}');
    }
  }

  lines.addAll([
    '--------------------------------',
    'Subtotal: CAD ${subtotal.toStringAsFixed(2)}',
  ]);

  if (discountAmount != null && discountAmount > 0) {
    lines.add('Discount: -CAD ${discountAmount.toStringAsFixed(2)}');
  }
  if (taxAmount != null && taxAmount > 0) {
    lines.add('Tax: CAD ${taxAmount.toStringAsFixed(2)}');
  }
  if (tip != null && tip > 0) {
    lines.add('Tip: CAD ${tip.toStringAsFixed(2)}');
  }

  lines.addAll([
    '================================',
    'TOTAL: CAD ${totalAmount.toStringAsFixed(2)}',
    '================================',
    'Thank you! Visit again',
    '',
  ]);

  await _printLines(lines);
}
